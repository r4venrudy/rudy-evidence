cmake_minimum_required(VERSION 3.24)
project(RudyEvidence VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(RE_BUILD_TESTS "Build tests" ${PROJECT_IS_TOP_LEVEL})
option(RE_BUILD_CLI "Build command line interface" ON)
option(RE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(RE_FUZZ "Build the Clang libFuzzer target" OFF)
option(RE_STATIC_RUNTIME "Link GNU/Clang Windows runtime statically" OFF)

add_library(re_blake3 STATIC vendor/blake3/blake3.c vendor/blake3/blake3_dispatch.c vendor/blake3/blake3_portable.c)
set_target_properties(re_blake3 PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
target_compile_definitions(re_blake3 PRIVATE BLAKE3_NO_AVX512 BLAKE3_NO_AVX2 BLAKE3_NO_SSE41 BLAKE3_NO_SSE2 BLAKE3_USE_NEON=0)
target_include_directories(re_blake3 PRIVATE vendor/blake3)
add_library(re_core STATIC src/evidence.cpp)
add_library(RudyEvidence::core ALIAS re_core)
set_target_properties(re_core PROPERTIES EXPORT_NAME core POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
target_compile_features(re_core PUBLIC cxx_std_20)
target_include_directories(re_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE vendor vendor/blake3)
target_link_libraries(re_core PRIVATE re_blake3)
add_library(re_capi SHARED src/c_api.cpp)
add_library(RudyEvidence::capi ALIAS re_capi)
set_target_properties(re_capi PROPERTIES EXPORT_NAME capi OUTPUT_NAME rudy_evidence VERSION ${PROJECT_VERSION} SOVERSION 1 CXX_VISIBILITY_PRESET hidden)
if(WIN32)
  set_target_properties(re_capi PROPERTIES PREFIX "")
endif()
target_link_libraries(re_capi PRIVATE re_core)
target_include_directories(re_capi PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set(re_targets re_core re_capi)
if(RE_BUILD_CLI)
  add_executable(rudy-evidence src/cli.cpp)
  target_include_directories(rudy-evidence PRIVATE vendor)
  target_link_libraries(rudy-evidence PRIVATE re_core)
  list(APPEND re_targets rudy-evidence)
endif()
foreach(t IN LISTS re_targets)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /WX /permissive- /utf-8)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(RE_SANITIZERS AND NOT MSVC)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PUBLIC -fsanitize=address,undefined)
  endif()
  if(RE_STATIC_RUNTIME AND MINGW)
    target_link_options(${t} PRIVATE -static)
  endif()
endforeach()
if(RE_SANITIZERS AND NOT MSVC)
  target_compile_options(re_blake3 PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
install(TARGETS re_core re_blake3 re_capi EXPORT RudyEvidenceTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(RE_BUILD_CLI)
  install(TARGETS rudy-evidence RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT RudyEvidenceTargets NAMESPACE RudyEvidence:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/RudyEvidence)
configure_package_config_file(cmake/RudyEvidenceConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/RudyEvidenceConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/RudyEvidence)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/RudyEvidenceConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/RudyEvidenceConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/RudyEvidenceConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/RudyEvidence)
install(FILES LICENSE THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/rudy-evidence)
install(DIRECTORY vendor/blake3/ DESTINATION ${CMAKE_INSTALL_DATADIR}/rudy-evidence/licenses/blake3 FILES_MATCHING PATTERN "LICENSE*")
install(FILES vendor/nlohmann/LICENSE.MIT DESTINATION ${CMAKE_INSTALL_DATADIR}/rudy-evidence/licenses/nlohmann)
if(RE_BUILD_TESTS)
  enable_testing()
  find_package(Threads REQUIRED)
  add_executable(core-tests tests/core.cpp)
  target_link_libraries(core-tests PRIVATE re_core Threads::Threads)
  add_test(NAME core-concurrency COMMAND core-tests)
  add_executable(c-api-tests tests/c_api.c)
  target_link_libraries(c-api-tests PRIVATE re_capi)
  add_test(NAME c-api-ownership COMMAND c-api-tests)
  if(RE_SANITIZERS AND NOT MSVC)
    # UBSan vptr instrumentation in the C++ shared library needs the C++ runtime.
    set_target_properties(c-api-tests PROPERTIES LINKER_LANGUAGE CXX)
    foreach(t core-tests c-api-tests)
      target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${t} PRIVATE -fsanitize=address,undefined)
    endforeach()
  endif()
  if(RE_BUILD_CLI AND NOT RE_SANITIZERS)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/contract.py $<TARGET_FILE:rudy-evidence> $<TARGET_FILE:re_capi>)
  endif()
endif()
if(RE_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "RE_FUZZ requires Clang")
  endif()
  target_compile_options(re_core PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  target_compile_options(re_blake3 PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  target_link_options(re_core PUBLIC -fsanitize=address,undefined)
  add_executable(evidence-fuzz tests/fuzz.cpp)
  target_link_libraries(evidence-fuzz PRIVATE re_core)
  target_compile_options(evidence-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(evidence-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
