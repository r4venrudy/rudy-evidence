cmake_minimum_required(VERSION 3.24)
project(EvidenceConsumer LANGUAGES C CXX)
find_package(RudyEvidence 0.1 CONFIG REQUIRED)
add_executable(c-consumer main.c)
target_link_libraries(c-consumer PRIVATE RudyEvidence::capi)
add_executable(cpp-consumer main.cpp)
target_link_libraries(cpp-consumer PRIVATE RudyEvidence::core)
enable_testing()
add_test(NAME installed-c COMMAND c-consumer)
add_test(NAME installed-cpp COMMAND cpp-consumer)
